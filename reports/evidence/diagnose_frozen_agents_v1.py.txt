"""Fresh paired-seat matches with immutable identities and trigger diagnostics.

No resume is supported: an existing output is an error. Local replay evidence
is generated by the real pinned interpreter with both agents reacting normally.
The diagnostic wrapper observes each agent's own input; it never gives either
agent the other player's private inventory or a hidden RNG seed.
"""
from __future__ import annotations
import argparse
import contextlib
import gzip
import hashlib
import importlib.metadata
import io
import json
import math
import statistics
import sys
import time
from concurrent.futures import ProcessPoolExecutor, as_completed
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def percentile(values, q):
    v = sorted(values)
    x = (len(v)-1)*q
    lo = int(x)
    return v[lo] + (v[min(lo+1,len(v)-1)]-v[lo])*(x-lo)

def summarize(games):
    valid = [g for g in games if g['valid']]
    margins = [g['margin'] for g in valid]
    if not valid:
        return {'valid_games':0,'invalid_games':len(games)}
    wins=sum(x>0 for x in margins);ties=sum(x==0 for x in margins)
    return {'valid_games':len(valid),'invalid_games':len(games)-len(valid),
            'wins':wins,'ties':ties,'losses':len(valid)-wins-ties,
            'points':(wins+.5*ties)/len(valid),'win_rate':wins/len(valid),
            'mean_margin':statistics.fmean(margins),'median_margin':statistics.median(margins),
            'p10_margin':percentile(margins,.1),'min_margin':min(margins),'max_margin':max(margins),
            'margin_variance':statistics.pvariance(margins),
            'elapsed_seconds':sum(g['elapsed_seconds'] for g in games)}

def play(job):
    candidate,opponent,seed,seat,trace_dir,expected = job
    with contextlib.redirect_stdout(io.StringIO()), contextlib.redirect_stderr(io.StringIO()):
        from kaggle_environments.agent import get_last_callable
        from src.current_env import make_environment
        paths = [candidate,opponent] if seat==0 else [opponent,candidate]
        assert [sha(p) for p in (candidate,opponent)] == expected
        funcs=[get_last_callable(Path(p).read_text(),path=p) for p in paths]
        logs=[[],[]]; digests=[hashlib.sha256(),hashlib.sha256()]
        times=[[],[]]; errors=[[],[]]; calls=[0,0]
        def wrapped(index):
            def act(obs,cfg):
                assert cfg.get('seed') is None, 'hidden seed leaked to agent'
                cfg['__raw_path__']=paths[index]
                start=time.perf_counter()
                action=funcs[index](obs,cfg)
                times[index].append(time.perf_counter()-start);calls[index]+=1
                digests[index].update(json.dumps(action,sort_keys=True,separators=(',',':')).encode()+b'\n')
                step=int(obs['step']);farm=obs['farms'][index];private=obs['private']
                animals={}; crops={}
                for row in farm['tiles']:
                    for tile in row:
                        if isinstance(tile,dict):
                            if tile.get('animal'):animals[tile['animal']]=animals.get(tile['animal'],0)+1
                            if tile.get('crop'):crops[tile['crop']]=crops.get(tile['crop'],0)+1
                wool=sum(int(inv.get('WOOL',0)) for inv in private['inventories'])
                if step%24==0 or (430<=step<=475) or (wool>=12 and obs['market']['prices']['WOOL']>=220):
                    impl=funcs[index].__globals__.get('_IMPL');route=None
                    if impl is not None:route=impl.chassis.players.get(index,{}).get('route')
                    logs[index].append({'step':step,'money':farm['money'],'animals':animals,'crops':crops,
                                        'positions':[farm['farmer'],*farm['hands']],
                                        'inventory':private['inventories'],'shed':private['shed'],
                                        'prices':obs['market']['prices'],'shops':obs['town']['unlocked_shops'],
                                        'route':route,'action':action})
                return action
            return act
        env=make_environment(seed)
        begin=time.perf_counter();env.run([wrapped(0),wrapped(1)]);elapsed=time.perf_counter()-begin
        statuses=[str(x.status) for x in env.state]
        last_step=int(env.state[0].observation.step)
        valid=statuses==['DONE','DONE'] and last_step==719 and len(env.steps)==720 and calls==[719,719]
        rewards=[float(x.reward or 0) for x in env.state]
        money=rewards[seat];other=rewards[1-seat]
        telemetry=[]
        for func in funcs:
            g=func.__globals__
            telemetry.append({k:v for k,v in g.items() if k in ('_E54_REPORT','_E63_REPORT','_E67_REPORT','_S1002_REPORT','_S793_REPORT','_HD2_REPORT')})
        result={'seed':seed,'candidate_seat':seat,'candidate_money':money,'opponent_money':other,
                'margin':money-other,'statuses':statuses,'last_step':last_step,'recorded_steps':len(env.steps),
                'calls':calls,'valid':valid,'elapsed_seconds':elapsed,'action_sha256':[d.hexdigest() for d in digests],
                'max_agent_seconds':[max(x or [0]) for x in times],
                'p99_agent_seconds':[percentile(x,.99) for x in times],
                'agent_seconds':[sum(x) for x in times],
                'telemetry':telemetry,'final_shops':list(env.state[0].observation.town.unlocked_shops),
                'source_sha256':expected,'structural_invalid_statuses':sum(s in ('INVALID','ERROR','TIMEOUT') for s in statuses),
                'semantic_noop_actions':'not measured; official interpreter silently ignores illegal operations'}
        if trace_dir:
            destination=Path(trace_dir);destination.mkdir(parents=True,exist_ok=True)
            trace_file=destination/f'{seed}_seat{seat}.json.gz'
            with gzip.open(trace_file,'wt') as f:json.dump({'result':result,'traces':logs},f,separators=(',',':'))
            result['trace_file']=str(trace_file.relative_to(ROOT))
        assert [sha(p) for p in (candidate,opponent)]==expected
        return result

def main():
    p=argparse.ArgumentParser();p.add_argument('candidate',type=Path);p.add_argument('opponent',type=Path)
    p.add_argument('--seeds',required=True);p.add_argument('--output',required=True,type=Path)
    p.add_argument('--workers',type=int,default=4);p.add_argument('--trace-dir',type=Path)
    a=p.parse_args();output=a.output.resolve()
    if output.exists():raise FileExistsError('Use a new output path; resume is deliberately disabled')
    candidate=a.candidate.resolve();opponent=a.opponent.resolve();seeds=[int(v) for v in a.seeds.split(',')]
    if len(seeds)!=len(set(seeds)):raise ValueError('Duplicate seeds')
    hashes=[sha(candidate),sha(opponent)]
    identity={'candidate':str(candidate.relative_to(ROOT)),'opponent':str(opponent.relative_to(ROOT)),
              'candidate_sha256':hashes[0],'opponent_sha256':hashes[1],
              'interpreter_sha256':sha(ROOT/'data/reference/master/kaggriculture.py'),
              'specification_sha256':sha(ROOT/'data/reference/master/kaggriculture.json'),
              'runner_sha256':sha(__file__),'environment_package':importlib.metadata.version('kaggle-environments'),
              'seed_hidden_from_agents':True,'shop_schedule':'unaltered real interpreter RNG',
              'opponent_mode':'independently reacting callable; no fixed action tape'}
    jobs=[(str(candidate),str(opponent),seed,seat,str(a.trace_dir.resolve()) if a.trace_dir else None,hashes) for seed in seeds for seat in (0,1)]
    report={'protocol':'fresh-paired-seats-pinned-official-interpreter-with-diagnostics-v1','identity':identity,
            'seeds':seeds,'expected_games':len(jobs),'complete':False,'games':[],'errors':[]}
    output.parent.mkdir(parents=True,exist_ok=True)
    def save():
        report['games'].sort(key=lambda g:(g['seed'],g['candidate_seat']))
        report['summary']=summarize(report['games'])
        temporary=output.with_suffix(output.suffix+'.tmp');temporary.write_text(json.dumps(report,indent=2)+'\n');temporary.replace(output)
    save()
    with ProcessPoolExecutor(max_workers=a.workers) as pool:
        futures={pool.submit(play,job):job for job in jobs}
        for future in as_completed(futures):
            job=futures[future]
            try:
                result=future.result();report['games'].append(result)
                if not result['valid']:report['errors'].append({'seed':job[2],'seat':job[3],'error':'incomplete or invalid match'})
                print(json.dumps({'finished':len(report['games']),'seed':job[2],'seat':job[3],'margin':result['margin'],'valid':result['valid']}),flush=True)
            except Exception as error:
                report['errors'].append({'seed':job[2],'seat':job[3],'error':repr(error)});print(report['errors'][-1],flush=True)
            save()
    report['complete']=len(report['games'])==len(jobs) and not report['errors'];save();print(json.dumps(report['summary'],indent=2))
    return 0 if report['complete'] else 1

if __name__=='__main__':raise SystemExit(main())
